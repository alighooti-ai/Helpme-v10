//====== TrendScore (shift=1، ATR-normalized) ======
bool ScoreVote(int emaH, int atrH, ENUM_TIMEFRAMES tf, double w, double &v, double &wo)
{
   double eb[1], ab[1];
   if(CopyBuffer(emaH,0,1,1,eb)!=1||eb[0]<=0) return false;
   if(CopyBuffer(atrH,0,1,1,ab)!=1||ab[0]<=0) return false;
   double cl=iClose(_Symbol,tf,1);
   double str=MathMin(1.0,MathAbs(cl-eb[0])/(ab[0]*InpScoreATRMult));
   v=w*((cl>eb[0])?1.0:-1.0)*str; wo=w; return true;
}
double CalcTrendScore()
{
   double vw=0,wt=0,v,w;
   if(ScoreVote(hEMA_H1, hATR_H1, PERIOD_H1, 4.0,v,w)){vw+=v;wt+=w;}
   if(ScoreVote(hEMA_M30,hATR_M30,PERIOD_M30,3.0,v,w)){vw+=v;wt+=w;}
   if(ScoreVote(hEMA_D1, hATR_D1, PERIOD_D1, 2.0,v,w)){vw+=v;wt+=w;}
   if(ScoreVote(hEMA_M5, hATR_M5, PERIOD_M5, 1.0,v,w)){vw+=v;wt+=w;}
   if(ScoreVote(hEMA_W1, hATR_W1, PERIOD_W1, 1.0,v,w)){vw+=v;wt+=w;}
   return (wt>0)?vw/wt*100.0:0.0;
}

//====== Hurst روی بازده (پنجره‌بندی غیرهمپوشان) ======
double CalcHurstH1(int bars)
{
   if(bars<40) return 0.5;
   double prices[];
   ArraySetAsSeries(prices,false);
   if(CopyClose(_Symbol,PERIOD_H1,1,bars+1,prices)<bars+1) return 0.5;
   double ret[]; ArrayResize(ret,bars);
   for(int i=0;i<bars;i++){
      if(prices[i]<=0||prices[i+1]<=0) return 0.5;
      ret[i]=MathLog(prices[i+1])-MathLog(prices[i]);
   }
   int lags[4]={5,10,20,40};
   double logRS[4],logL[4]; int vp=0;
   for(int li=0;li<4;li++){
      int lag=lags[li],nw=bars/lag;
      if(nw<2) continue;
      double sRS=0; int cRS=0;
      for(int wi=0;wi<nw;wi++){
         int b=wi*lag; double mn=0;
         for(int i=0;i<lag;i++) mn+=ret[b+i]; mn/=lag;
         double cum=0,mx=0,mi=0;
         for(int i=0;i<lag;i++){
            cum+=(ret[b+i]-mn);
            if(i==0){mx=cum;mi=cum;}
            else{if(cum>mx)mx=cum;if(cum<mi)mi=cum;}
         }
         double rng=mx-mi,sd=0;
         for(int i=0;i<lag;i++) sd+=MathPow(ret[b+i]-mn,2);
         sd=MathSqrt(sd/lag);
         if(sd>0&&rng>0){sRS+=rng/sd;cRS++;}
      }
      if(cRS==0) continue;
      logRS[vp]=MathLog(sRS/cRS); logL[vp]=MathLog((double)lag); vp++;
   }
   if(vp<2) return 0.5;
   double sx=0,sy=0,sxy=0,sx2=0;
   for(int i=0;i<vp;i++){sx+=logL[i];sy+=logRS[i];sxy+=logL[i]*logRS[i];sx2+=logL[i]*logL[i];}
   double n=vp,dn=(n*sx2-sx*sx);
   if(dn==0) return 0.5;
   double H=(n*sxy-sx*sy)/dn;
   return MathMax(0.0,MathMin(1.0,H));
}

double CalcSpikeRatioH1()
{
   double atr[1];
   if(CopyBuffer(hATR_H1,0,1,1,atr)!=1||atr[0]<=0) return 1.0;
   return (iHigh(_Symbol,PERIOD_H1,1)-iLow(_Symbol,PERIOD_H1,1))/atr[0];
}
double GetATR_H1()
{ double a[1]; return (CopyBuffer(hATR_H1,0,1,1,a)==1)?a[0]:0.0; }
double GetADX_D1()
{ double a[1]; return (CopyBuffer(hADX_D1,0,1,1,a)==1)?a[0]:0.0; }

//====== V7: VolRatio regime — ATR سریع/کند (جایگزین ADX برای InpRegimeMode=1) ======
//====== ایدهٔ خودِ مانیفست اصلی (MktPhase/ADR ratio) که هیچ‌وقت ساخته نشد؛ اینجا با ATR ساده‌تر پیاده شد ======
double CalcVolRatio()
{
   double f[1],s[1];
   if(CopyBuffer(hATR_VR_Fast,0,1,1,f)!=1||f[0]<=0) return 1.0;
   if(CopyBuffer(hATR_VR_Slow,0,1,1,s)!=1||s[0]<=0) return 1.0;
   return f[0]/s[0];
}

//====== V7: CUSUM دوطرفه (Page-Hinkley روی بازدهٔ H1) — جایگزین ADX برای InpRegimeMode=2 ======
//====== k = slack (نویز مورد قبول)، h = آستانهٔ تشخیص تغییر. هر دو بر حسب ATR(H1) مقیاس می‌شوند ======
void UpdateCUSUM(double closeP, double atr)
{
   if(!g_cusumInit || atr<=0){ g_lastCloseForCUSUM=closeP; g_cusumInit=true; return; }
   double ret = closeP - g_lastCloseForCUSUM;
   g_lastCloseForCUSUM = closeP;
   double k = InpCUSUM_K_ATRMult*atr;
   g_cusumPos = MathMax(0.0, g_cusumPos + ret - k);
   g_cusumNeg = MathMin(0.0, g_cusumNeg + ret + k);
}
bool CUSUM_ChangeDetected(double atr)
{
   double h = InpCUSUM_H_ATRMult*atr;
   bool hit = (g_cusumPos > h) || (g_cusumNeg < -h);
   if(hit){ g_cusumPos=0; g_cusumNeg=0; }  // ریست بعد از تشخیص، طبق تعریف استاندارد CUSUM
   return hit;
}

//====== V7: Kalman Filter — مدل Local-Linear-Trend (level+slope)، کوواریانس ۲x۲ کامل ======
//====== فقط Shadow-Log می‌شود؛ روی هیچ تصمیم معاملاتی در V7 اثر نمی‌گذارد ======
void UpdateKalman(double closeP)
{
   if(!g_kalmanInit)
   {
      g_kalmanLevel=closeP; g_kalmanTrend=0;
      g_kalmanP11=1; g_kalmanP12=0; g_kalmanP21=0; g_kalmanP22=1;
      g_kalmanInit=true;
      return;
   }
   // --- Predict ---
   double levelPred = g_kalmanLevel + g_kalmanTrend;
   double trendPred = g_kalmanTrend;
   double p11 = g_kalmanP11 + g_kalmanP12 + g_kalmanP21 + g_kalmanP22 + InpKalmanQLevel;
   double p12 = g_kalmanP12 + g_kalmanP22;
   double p21 = g_kalmanP21 + g_kalmanP22;
   double p22 = g_kalmanP22 + InpKalmanQTrend;
   // --- Update (measurement = closeP) ---
   double y = closeP - levelPred;
   double S = p11 + InpKalmanR;
   if(S<=0) S=0.0000001;
   double K1 = p11/S;
   double K2 = p21/S;
   g_kalmanLevel = levelPred + K1*y;
   g_kalmanTrend = trendPred + K2*y;
   g_kalmanP11 = (1-K1)*p11;
   g_kalmanP12 = (1-K1)*p12;
   g_kalmanP21 = p21 - K2*p11;
   g_kalmanP22 = p22 - K2*p12;
}

//====== Swing High/Low: فرکتال ۵-کندلی روی D1 (شکل استاندارد Bill Williams) ======
//====== آخرین سقف/کف تأییدشده را برمی‌گرداند — فقط دو کندل بعد از تشکیل تأیید می‌شود ======
void ScanSwingPoints()
{
   int bars = InpSwingScanBars;
   double highs[], lows[];
   ArraySetAsSeries(highs,false); ArraySetAsSeries(lows,false);
   if(CopyHigh(_Symbol,PERIOD_D1,1,bars,highs) < bars) return;
   if(CopyLow(_Symbol,PERIOD_D1,1,bars,lows)  < bars) return;
   int n = bars;

   for(int i=n-3; i>=2; i--)
   {
      bool isHigh=true;
      for(int k=i-2;k<=i+2;k++){ if(k==i) continue; if(highs[k]>=highs[i]){isHigh=false;break;} }
      if(isHigh){ g_lastSwingHigh = highs[i]; break; }
   }
   for(int i=n-3; i>=2; i--)
   {
      bool isLow=true;
      for(int k=i-2;k<=i+2;k++){ if(k==i) continue; if(lows[k]<=lows[i]){isLow=false;break;} }
      if(isLow){ g_lastSwingLow = lows[i]; break; }
   }
}

{
   ulong dt=trade.ResultDeal();
   if(dt==0||!HistoryDealSelect(dt)) return 0.0;
   return HistoryDealGetDouble(dt,DEAL_PROFIT)
         +HistoryDealGetDouble(dt,DEAL_SWAP)
         +HistoryDealGetDouble(dt,DEAL_COMMISSION);
}

//====== سود شناور فعلی یک پوزیشن باز (شامل سواپ) — برای Net-Close ======

   if(!PosExists(ticket)) return 0.0;
   return PositionGetDouble(POSITION_PROFIT) + PositionGetDouble(POSITION_SWAP);
}

             ev,
             DoubleToString(pnl,2),
             DoubleToString(AccountInfoDouble(ACCOUNT_BALANCE),2),
             DoubleToString(AccountInfoDouble(ACCOUNT_EQUITY),2),
             // V7 additions
             DoubleToString(g_curClose,_Digits),
             g_gateScoreCond?1:0, g_gateHoldOK?1:0, g_gateCoolOK?1:0, g_gateHurstOK?1:0,
             g_gateRegimeOK_ADX?1:0, g_gateRegimeOK_VolRatio?1:0, g_gateRegimeOK_CUSUM?1:0, g_gateStructOK?1:0,
             DoubleToString(g_curVolRatio,3), DoubleToString(g_cusumPos,5), DoubleToString(g_cusumNeg,5),
             DoubleToString(g_kalmanLevel,_Digits), DoubleToString(g_kalmanTrend,_Digits+2),
             InpRegimeMode, InpPersistenceMode);
   g_eventThisBar = true;
}

// باز کردن پای روند تازه در جهت مشخص. reference قیمت برای trailing/open ثبت می‌شود.
bool OpenFreshTrend(int dir)
{
   trade.SetExpertMagicNumber(InpMagicTrend);
   bool ok=(dir==1)?trade.Buy(InpTrendLegVolume,_Symbol,0,0,0,"AFA_TREND")
                   :trade.Sell(InpTrendLegVolume,_Symbol,0,0,0,"AFA_TREND");
   if(ok)
   {
      g_trendTicket = trade.ResultOrder();
      g_trendDir    = dir;
      double refP   = (dir==1)?SymbolInfoDouble(_Symbol,SYMBOL_ASK):SymbolInfoDouble(_Symbol,SYMBOL_BID);
      g_trendOpenPrice = refP;
      g_trendPeakPrice = refP;
   }
   else { g_trendTicket=0; g_trendDir=0; }
   return ok;
}

//+------------------------------------------------------------------+
void OnTick()
{
   datetime curBar=iTime(_Symbol,PERIOD_H1,0);
   if(curBar==g_lastBarTime) return;
   g_lastBarTime=curBar;
   g_eventThisBar=false;

   if(g_trendDir!=0) g_holdBarCount++;
   if(g_cooldownCounter>0) g_cooldownCounter--;

   // یک بار در روز، ساختار بازار (Swing High/Low) را به‌روز کن
   datetime curD1=iTime(_Symbol,PERIOD_D1,0);
   if(curD1!=g_lastD1Time){ g_lastD1Time=curD1; ScanSwingPoints(); }

   double ts   = CalcTrendScore();
   double h    = CalcHurstH1(InpHurstBars);
   double atr  = GetATR_H1();
   double sr   = CalcSpikeRatioH1();
   double adx  = GetADX_D1();
   double closeP = iClose(_Symbol,PERIOD_H1,1);   // مرجع قیمتی عمومی برای ساختار/تریلینگ (کندل بسته)
   bool   spike    = (sr>=InpSpikeATRRatio);
   bool   spikeBlk = (sr>=InpMaxSpikeRatioBlock);

   //===== V7: سه کاندید Regime + Kalman — هرکدام جدا محاسبه و لاگ می‌شوند تا با یک بک‌تست =====
   //===== بشه سه‌تا را منصفانه مقایسه کرد. فقط InpRegimeMode تعیین می‌کند کدام واقعاً استفاده شود =====
   g_curClose    = closeP;
   g_curVolRatio = CalcVolRatio();
   UpdateCUSUM(closeP, atr);
   if(CUSUM_ChangeDetected(atr)) g_cusumLatchCounter = InpCUSUM_LatchBars;
   else if(g_cusumLatchCounter>0) g_cusumLatchCounter--;
   if(InpKalmanShadowLog) UpdateKalman(closeP);

   g_gateRegimeOK_ADX      = (!InpEnableRegimeFilter) || (adx >= InpRegimeADXMin);
   g_gateRegimeOK_VolRatio = (!InpEnableRegimeFilter) || (g_curVolRatio >= InpVolRatioThreshold);
   g_gateRegimeOK_CUSUM    = (!InpEnableRegimeFilter) || (g_cusumLatchCounter > 0);

   bool regimeOK;
   if(InpRegimeMode==1)      regimeOK = g_gateRegimeOK_VolRatio;
   else if(InpRegimeMode==2) regimeOK = g_gateRegimeOK_CUSUM;
   else                       regimeOK = g_gateRegimeOK_ADX;   // پیش‌فرض = دقیقاً رفتار V6، بدون تغییر

   if(g_trendTicket!=0&&!PosExists(g_trendTicket)){g_trendTicket=0;g_trendDir=0;}
   if(g_hedgeTicket!=0&&!PosExists(g_hedgeTicket)) g_hedgeTicket=0;

   //===== STEP 1: بدون پای روند -> منتظر ورود اولیه =====
   if(g_trendDir==0)
   {
      bool structOK = true;
      int  wantDir  = (ts>0)?1:-1;
      if(InpEnableStructureFilter)
      {
         if(wantDir==1) structOK = (closeP > g_lastSwingHigh && g_lastSwingHigh>0);
         else            structOK = (closeP < g_lastSwingLow  && g_lastSwingLow>0);
      }
      g_gateStructOK = structOK;
      if(!spikeBlk && regimeOK && structOK && MathAbs(ts)>=InpInitTrendThreshold)
      {
         bool opened = OpenFreshTrend(wantDir);
         if(opened)
         {
            ResetAfterReversal();
            LogRow("OPEN_TREND",ts,h,atr,sr,adx);
         }
         else
         {
            // V7 فیکس: قبلاً اینجا بدون شرط LogRow("OPEN_TREND"..) صدا زده می‌شد حتی وقتی
            // trade.Buy/Sell رد می‌شد (TrendTicket صفر می‌ماند) — یک لاگ فانتوم که تعداد
            // معاملات واقعی را غلط نشان می‌داد (۲ نمونهٔ مستند در لاگ V6). حالا سفارش ناموفق
            // با یک Event صریح و جدا ثبت می‌شود، نه به اسم OPEN_TREND.
            LogRow("OPEN_TREND_FAILED_ORDER_REJECTED",ts,h,atr,sr,adx);
         }
      }
      else LogRow("IDLE_WAIT_TREND",ts,h,atr,sr,adx);
      return;
   }

   //===== STEP 1.6: سقف ضرر مستقل (V6) — همچنان اولین اولویت =====
   // عمداً همیشه اول می‌ماند: این یک کف امن مستقل برای ورودی‌های اساساً غلط است،
   // نه رقیب Reversal/Trailing — طبق طراحی اصلی‌اش نباید منتظر آن‌ها بماند.
   if(InpEnableMaxAdverse && atr>0)
   {
      double adverseNow = (g_trendDir==1) ? (g_trendOpenPrice-closeP) : (closeP-g_trendOpenPrice);
      if(adverseNow >= InpMaxAdverseATR*atr)
      {
         double hp=0,tp=0;
         if(g_hedgeTicket!=0 && PosExists(g_hedgeTicket)){ trade.PositionClose(g_hedgeTicket); hp=GetLastDealPnL(); }
         if(PosExists(g_trendTicket)){ trade.PositionClose(g_trendTicket); tp=GetLastDealPnL(); }
         LogRow("MAX_ADVERSE_STOP",ts,h,atr,sr,adx,tp+hp);
         g_trendTicket=0; g_hedgeTicket=0; g_trendDir=0;
         ResetAfterReversal();
         return;
      }
   }

   //===== STEP 2 (V7: حالا قبل از Trailing): بررسی Reversal-Swap =====
   // باگ V6 که این‌جا فیکس شد: Trailing Stop قبلاً همین‌جا زودتر return می‌زد، پس این بخش
   // در تنظیم کامل (Hedge+Trailing روشن) هیچ‌وقت اجرا نمی‌شد — ۰ از ۳۷ بستنِ پای روند در
   // ۷ ماه لاگ واقعی از مسیر Reversal بود. حالا Reversal همیشه اول فرصت اجرا دارد.
   double scoreAgainst=(g_trendDir==1)?-ts:ts;
   g_gateScoreCond = (scoreAgainst >= InpReversalThreshold);

   if(InpPersistenceMode==1)
   {
      // V7 حالت decay: یک کندل نویزی فقط یک پله عقب می‌برد، نه کل پیشرفت را صفر می‌کند.
      // نمونهٔ مستند از لاگ واقعی: ۱۱ کندل پیشرفت با یک کندل نویزی صفر شد و ۳ روز تاخیر اضافه کرد؛
      // با decay همان کندل نویزی شمارنده را فقط به ۱۰ می‌رساند.
      if(g_gateScoreCond) g_reversalCounter = g_reversalCounter+1;
      else                 g_reversalCounter = MathMax(0, g_reversalCounter-InpPersistDecayStep);
   }
   else
   {
      // V6 legacy: ریست سخت با هر کندل ناموفق — رفتار پیش‌فرض بدون تغییر تا با decay مقایسه شود
      if(g_gateScoreCond) g_reversalCounter++;
      else                 g_reversalCounter=0;
   }

   g_gateHoldOK  = (g_holdBarCount >= InpMinHoldBars);
   g_gateCoolOK  = (g_cooldownCounter == 0);
   // V9: Hurst از گیت Reversal حذف شد. طبق جدول §۲ مانیفست اصلی، نقش Hurst فقط
   // «فیلتر شرایط باز کردن پای هج» است؛ در کل مانیفست، Hurst هیچ‌جا برای Reversal-Swap
   // ذکر نشده. در V6/V7 اشتباهاً هم‌زمان برای هج و Reversal استفاده می‌شد — دادهٔ لاگ واقعی
   // نشون داد دقیقاً همون لحظاتی که ReversalCounter آماده بود (۹ از ۱۴ مورد)، Hurst رد می‌کرد،
   // چون یک حرکت قوی و پایدار (Hurst بالا) دقیقاً همون چیزیه که سیگنال Reversal رو می‌سازه —
   // این گیت داشت با علت خودش می‌جنگید. g_gateHurstOK همچنان محاسبه و لاگ می‌شه (برای دیباگ)
   // ولی دیگه در doReversal شرط نیست.
   g_gateHurstOK = (h < InpReversalHurstMax);
   // V8_AI2: قبلاً این بلوک از همون InpEnableStructureFilter ورود اولیه استفاده می‌کرد —
   // که طبق لاگ مستند باعث می‌شد Reversal در کل ۷ ماه صفر بار فایر بشه (StructOK فقط ۲٪ True).
   // حالا مستقل شده: InpReversalRequireStruct (دیفالت false) این گیت رو کنترل می‌کنه،
   // بدون این‌که به کیفیت ورود اولیه (که InpEnableStructureFilter هنوز جدا کنترلش می‌کنه) دست بزنه.
   bool structOKRev = true;
   if(InpReversalRequireStruct)
   {
      // شکست ساختار واقعی: در روند صعودی، فقط وقتی کف نوسان قبلی شکسته شود اجازهٔ Reversal بده؛ در نزولی، فقط با شکست سقف
      if(g_trendDir==1) structOKRev = (closeP < g_lastSwingLow  && g_lastSwingLow>0);
      else               structOKRev = (closeP > g_lastSwingHigh && g_lastSwingHigh>0);
   }
   g_gateStructOK = structOKRev;

   bool doReversal = (g_reversalCounter >= InpReversalPersistBars)
                     && g_gateHoldOK && g_gateCoolOK && regimeOK && structOKRev;

   if(doReversal)
   {
      int newDir=-g_trendDir;
      double tpnl=0.0;
      if(g_trendTicket!=0 && PosExists(g_trendTicket))
      { trade.PositionClose(g_trendTicket); tpnl=GetLastDealPnL(); }
      LogRow("CLOSE_TREND_ON_REVERSAL",ts,h,atr,sr,adx,tpnl);
      g_trendTicket=0;

      if(g_hedgeTicket!=0 && PosExists(g_hedgeTicket))
      {
         // V9 — Role-Swap واقعی طبق §۱ مانیفست: "پای هج نقش پای روند جدید را می‌گیرد".
         // برخلاف V7/V8 که پای هج رو می‌بست و یک پای روند تازه باز می‌کرد (اسپرد دوبار،
         // و اصلاً همون مکانیزمی نیست که مانیفست می‌خواد)، اینجا پوزیشن هج موجود — که از
         // قبل دقیقاً در جهت جدید باز بود — بدون بستن/بازکردن، به‌عنوان پای روند جدید
         // شناخته می‌شه. توجه: مجیک‌نامبر پوزیشن در ترمینال همچنان InpMagicHedge می‌مونه
         // (در MQL5 غیرقابل تغییره)، ولی EA داخلی از این به بعد این تیکت رو trend می‌دونه.
         g_trendTicket    = g_hedgeTicket;
         g_trendDir       = newDir;
         g_trendOpenPrice = PositionGetDouble(POSITION_PRICE_OPEN);
         g_trendPeakPrice = g_trendOpenPrice;
         g_hedgeTicket    = 0;
         g_holdBarCount   = 0;
         g_reversalCounter=0;
         g_cooldownCounter=InpReversalCooldownBars;
         LogRow("HEDGE_PROMOTED_TO_TREND",ts,h,atr,sr,adx);
      }
      else
      {
         // فال‌بک: اگه لحظهٔ Reversal هج بازی نداشتیم (اکثر مواقع در دادهٔ واقعی — فقط
         // ۶-۷ بار در ۷ ماه هج باز شده)، طبق منطق قبلی یک پای روند تازه باز کن.
         bool opened2 = OpenFreshTrend(newDir);
         ResetAfterReversal();
         if(opened2) LogRow("OPEN_TREND_AFTER_REVERSAL",ts,h,atr,sr,adx);
         else         LogRow("OPEN_TREND_AFTER_REVERSAL_FAILED_ORDER_REJECTED",ts,h,atr,sr,adx);
      }
      return;
   }

   //===== STEP 1.5 (V7: حالا بعد از Reversal): به‌روزرسانی قله + بررسی Trailing Stop =====
   // فقط وقتی به این‌جا می‌رسیم که Reversal این کندل نزده — یعنی حالا Trailing واقعاً یک
   // fallback است، نه رقیب همیشه‌-برنده.
   if(g_trendDir==1) g_trendPeakPrice = MathMax(g_trendPeakPrice, closeP);
   else               g_trendPeakPrice = MathMin(g_trendPeakPrice, closeP);

   if(InpEnableTrailingStop)
   {
      double peakProfit = (g_trendDir==1) ? (g_trendPeakPrice-g_trendOpenPrice) : (g_trendOpenPrice-g_trendPeakPrice);
      double giveBack   = (g_trendDir==1) ? (g_trendPeakPrice-closeP)           : (closeP-g_trendPeakPrice);
      bool wasProfitable = (atr>0) && (peakProfit >= InpTrailingMinProfitATR*atr);
      bool trailHit       = wasProfitable && (atr>0) && (giveBack >= InpTrailingATRMult*atr);
      if(trailHit)
      {
         double hp=0,tp=0;
         if(g_hedgeTicket!=0 && PosExists(g_hedgeTicket)){ trade.PositionClose(g_hedgeTicket); hp=GetLastDealPnL(); }
         if(PosExists(g_trendTicket)){ trade.PositionClose(g_trendTicket); tp=GetLastDealPnL(); }
         LogRow("TRAILING_STOP_TREND",ts,h,atr,sr,adx,tp+hp);
         g_trendTicket=0; g_hedgeTicket=0; g_trendDir=0;
         ResetAfterReversal();
         return;
      }
   }

   if(!InpEnableHedge)
   { if(!g_eventThisBar) LogRow("BAR_TICK",ts,h,atr,sr,adx); return; }

   //===== STEP 3: مدیریت پای هج =====
   double curPrice=(g_trendDir==1)?SymbolInfoDouble(_Symbol,SYMBOL_BID):SymbolInfoDouble(_Symbol,SYMBOL_ASK);

   // --- STEP 3a: Net-Close — اولویت اول، قبل از هر منطق دیگر بستن هج ---
   if(InpEnableNetClose && g_hedgeTicket!=0)
   {
      double combined = GetFloatingPnL(g_trendTicket) + GetFloatingPnL(g_hedgeTicket);
      if(combined >= InpNetCloseTargetProfit)
      {
         double hp=0,tp=0;
         if(PosExists(g_hedgeTicket)){ trade.PositionClose(g_hedgeTicket); hp=GetLastDealPnL(); }
         if(PosExists(g_trendTicket)){ trade.PositionClose(g_trendTicket); tp=GetLastDealPnL(); }
         g_hedgeTicket=0; g_trendTicket=0;
         LogRow("NET_CLOSE_BOTH",ts,h,atr,sr,adx,tp+hp);

         // بلافاصله پای روند تازه هم‌جهت (نه معکوس) باز کن — این یک Reversal نیست، فقط ریست هزینه است
         OpenFreshTrend(g_trendDir==0?1:g_trendDir); // g_trendDir توسط OpenFreshTrend عوض نمی‌شود مگر شکست بخورد؛ جهت قبلی حفظ می‌شود
         ResetHedgeCycleState();
         LogRow("REOPEN_TREND_AFTER_NETCLOSE",ts,h,atr,sr,adx);
         return;
      }
   }

   if(g_hedgeTicket==0)
   {
      // --- شرایط باز کردن هج ---
      double refPrice;
      if(g_hedgeOpenEver) refPrice=g_lastHedgeClosePrice;
      else if(PosExists(g_trendTicket)) refPrice=PositionGetDouble(POSITION_PRICE_OPEN);
      else refPrice=curPrice;

      double advMove=(g_trendDir==1)?(refPrice-curPrice):(curPrice-refPrice);
      bool mOK  = advMove>=InpHedgeOpenATRMult*atr;
      bool hOK  = h<InpHurstThreshold;
      bool sOK  = !spike && !spikeBlk;

      bool monoOK=true;
      if(g_hedgeOpenEver)
      {
         if(g_trendDir==1) monoOK=(curPrice>g_lastHedgeEntryPrice);
         else               monoOK=(curPrice<g_lastHedgeEntryPrice);
      }

      if(mOK&&hOK&&sOK&&monoOK)
      {
         int hdir=-g_trendDir;
         trade.SetExpertMagicNumber(InpMagicHedge);
         bool ok=(hdir==1)?trade.Buy(InpHedgeLegVolume,_Symbol,0,0,0,"AFA_HEDGE")
                          :trade.Sell(InpHedgeLegVolume,_Symbol,0,0,0,"AFA_HEDGE");
         trade.SetExpertMagicNumber(InpMagicTrend);
         if(ok){
            g_hedgeTicket=trade.ResultOrder();
            g_lastHedgeEntryPrice=curPrice;
            g_hedgeOpenEver=true;
            g_hedgeAdverseExtreme=curPrice;
            LogRow("OPEN_HEDGE",ts,h,atr,sr,adx);
         }
      }
   }
   else
   {
      // --- STEP 3b: fallback بستن هج (فقط اگر Net-Close هنوز نرسیده) ---
      if(g_trendDir==1) g_hedgeAdverseExtreme=MathMin(g_hedgeAdverseExtreme,curPrice);
      else               g_hedgeAdverseExtreme=MathMax(g_hedgeAdverseExtreme,curPrice);

      double advT=(g_trendDir==1)?(g_lastHedgeEntryPrice-g_hedgeAdverseExtreme)
                                  :(g_hedgeAdverseExtreme-g_lastHedgeEntryPrice);
      double retF=(g_trendDir==1)?(curPrice-g_hedgeAdverseExtreme)
                                  :(g_hedgeAdverseExtreme-curPrice);
      bool rOK   =(advT>0)&&(retF>=InpHedgeCloseRetraceP*advT);
      bool trOK  =((g_trendDir==1&&ts>=InpInitTrendThreshold)||(g_trendDir==-1&&-ts>=InpInitTrendThreshold));
      bool spSafe=spike;

      if(rOK||trOK||spSafe)
      {
         double hp=0.0;
         if(PosExists(g_hedgeTicket)){trade.PositionClose(g_hedgeTicket);hp=GetLastDealPnL();}
         g_lastHedgeClosePrice=curPrice; g_hedgeTicket=0; g_hedgeAdverseExtreme=0.0;
         string ev=rOK?"CLOSE_HEDGE_RETRACE":(trOK?"CLOSE_HEDGE_TREND_RESUMED":"CLOSE_HEDGE_SPIKE_SAFETY");
         LogRow(ev,ts,h,atr,sr,adx,hp);
      }
   }

   if(!g_eventThisBar) LogRow("BAR_TICK",ts,h,atr,sr,adx);
}
